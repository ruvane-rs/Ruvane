use ruvane::{Attribute, Html, InternalUrl, Page, PageContext, Tag, element, fragment, link};

pub fn stylesheet() -> InternalUrl {
    InternalUrl::new("/assets/site.css").expect("fixed stylesheet path")
}

pub fn site_layout(context: &PageContext<'_>, title: &str, children: Html) -> Html {
    let navigation = element(Tag::Nav).child(fragment([
        link(&context.url("home", &[]).expect("registered home"), "Home"),
        link(
            &context.url("about", &[]).expect("registered about"),
            "About",
        ),
        link(
            &context
                .url("hello", &[("name", "rust")])
                .expect("registered greeting"),
            "Meet Rust",
        ),
    ]));
    element(Tag::Main)
        .child(navigation)
        .child(element(Tag::H1).child(title))
        .child(children)
        .render()
}

pub fn greeting_layout(_: &PageContext<'_>, _: &str, children: Html) -> Html {
    element(Tag::Section)
        .attr(Attribute::Class, "greeting")
        .child(children)
        .render()
}

pub fn page(context: &PageContext<'_>, title: &str, body: impl Into<Html>) -> Page {
    Page::new(title, body)
        .metadata(
            context
                .metadata("A small site built with Ruvane.")
                .expect("fixed description"),
        )
        .stylesheet(stylesheet())
}
