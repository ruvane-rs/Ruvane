use crate::{components, pages, store};
use ruvane::{ClientPageData, FilePage};
pub fn page() -> FilePage {
    FilePage::client("settings", components::SETTINGS, pages::zero, |context| {
        Ok(
            ClientPageData::new("Task settings / Ruvane", store::snapshot()?)
                .metadata(pages::metadata(context, "Configure the task workspace.")?),
        )
    })
}
