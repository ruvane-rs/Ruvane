use crate::{components, pages, store};
use ruvane::{ClientPageData, FilePage};
pub fn page() -> FilePage {
    FilePage::client("home", components::HOME, pages::zero, |context| {
        Ok(
            ClientPageData::new("Home / Ruvane", store::snapshot()?).metadata(pages::metadata(
                context,
                "Shared Rust pages and server actions.",
            )?),
        )
    })
}
