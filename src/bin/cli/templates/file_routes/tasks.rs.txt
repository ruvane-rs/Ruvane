use crate::{components, pages, store};
use ruvane::{ClientPageData, FilePage};
pub fn page() -> FilePage {
    FilePage::client("tasks", components::TASKS, pages::zero, |context| {
        Ok(ClientPageData::new("Tasks / Ruvane", store::snapshot()?)
            .fallback(pages::native_forms(None)?)
            .metadata(pages::metadata(
                context,
                "Manage tasks and follow server updates.",
            )?))
    })
}
