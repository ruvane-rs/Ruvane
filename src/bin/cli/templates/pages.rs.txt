use crate::components;
use ruvane::{
    App, AppError, MediaType, Page, PageMode, PageParams, PageRoute, Router, SiteOrigin,
    StaticSite, Tag, element,
};

const CSS: &[u8] = include_bytes!("../assets/site.css");

pub fn application() -> Result<App, AppError> {
    let articles = crate::content::collection().map_err(|_| AppError::InvalidPage)?;
    let home_articles: Vec<_> = articles
        .iter()
        .map(|article| {
            (
                article.slug().as_str().to_owned(),
                article.title().to_owned(),
            )
        })
        .collect();
    let mut app = App::new();
    app.layout("site", None, components::site_layout)?;
    app.layout("greeting", Some("site"), components::greeting_layout)?;
    app.page(
        PageRoute::new("home", "/")?,
        Some("site"),
        PageMode::fixed(),
        move |context| {
            Ok(components::page(
                context,
                "A small site. Your next idea.",
                element(Tag::Section)
                    .child(element(Tag::P).child("Welcome. Make something worth sharing."))
                    .child(
                        element(Tag::Ul).child(ruvane::fragment(home_articles.iter().map(
                            |(slug, title)| {
                                element(Tag::Li)
                                    .child(ruvane::link(
                                        &context
                                            .url("article", &[("slug", slug.as_str())])
                                            .expect("registered article"),
                                        title.as_str(),
                                    ))
                                    .render()
                            },
                        ))),
                    ),
            ))
        },
    )?;
    app.page(
        PageRoute::new("about", "/about/")?,
        Some("site"),
        PageMode::fixed(),
        |context| {
            Ok(components::page(
                context,
                "About",
                element(Tag::P).child("Clear ideas, thoughtful pages, and room to explore."),
            ))
        },
    )?;
    app.page(
        PageRoute::new("hello", "/hello/{name}/")?,
        Some("greeting"),
        PageMode::Static(vec![
            PageParams::new(&[("name", "rust")])?,
            PageParams::new(&[("name", "web")])?,
        ]),
        |context| {
            let name = context.param("name").unwrap_or("world");
            Ok(components::page(
                context,
                "Hello",
                element(Tag::P).child(format!("Hello, {name}!")),
            ))
        },
    )?;
    app.content_pages(
        PageRoute::new("article", "/articles/{slug}/")?,
        Some("site"),
        articles,
        |context, article| {
            Ok(Page::new(article.title(), article.body().clone())
                .metadata(
                    context
                        .metadata(article.description())
                        .map_err(|_| ruvane::PageError::InternalServerError)?,
                )
                .stylesheet(components::stylesheet()))
        },
    )?;
    app.asset("/assets/site.css", CSS, MediaType::Css)?;
    app.not_found(Some("site"), || {
        Page::new(
            "Page not found",
            element(Tag::P).child("That page is not available."),
        )
        .stylesheet(components::stylesheet())
    })?;
    app.internal_error(Some("site"), || {
        Page::new(
            "Something went wrong",
            element(Tag::P).child("Please try again later."),
        )
        .stylesheet(components::stylesheet())
    })?;
    Ok(app)
}

pub fn router() -> Result<Router, AppError> {
    application()?.router(None)
}

pub fn site(origin: &SiteOrigin) -> Result<StaticSite, AppError> {
    let app = application()?;
    for name in app.server_only_routes() {
        println!("Server-only page omitted from static export: {name}");
    }
    app.static_site(origin)
}
