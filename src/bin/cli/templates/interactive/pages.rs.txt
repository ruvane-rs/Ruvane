use crate::{components, store};
use ruvane::{
    App, ClientPageData, ComponentProps, Html, InitialState, InternalUrl, NavigationMatch,
    PageContext, PageError, PageRoute, PostForm, Tag, element,
};
fn zero(_: &NavigationMatch) -> Result<InitialState, PageError> {
    Ok(InitialState::Number(0))
}
fn task(route: &NavigationMatch) -> Result<InitialState, PageError> {
    let raw = route.param("id").ok_or(PageError::BadRequest)?;
    let id = raw.parse::<u32>().map_err(|_| PageError::BadRequest)?;
    if id == 0 || id.to_string() != raw {
        return Err(PageError::BadRequest);
    }
    Ok(InitialState::Number(id))
}
pub fn application() -> App {
    let mut app = App::new();
    app.components(components::create_component, components::ERROR)
        .unwrap();
    app.client_page(
        PageRoute::new("home", "/").unwrap(),
        components::HOME,
        zero,
        |_| Ok(ClientPageData::new("Home / Ruvane", store::snapshot()?)),
    )
    .unwrap();
    app.client_page(
        PageRoute::new("tasks", "/tasks").unwrap(),
        components::TASKS,
        zero,
        |_| {
            Ok(ClientPageData::new("Tasks / Ruvane", store::snapshot()?)
                .fallback(native_forms(None)?))
        },
    )
    .unwrap();
    app.client_page(
        PageRoute::new("task", "/tasks/{id}").unwrap(),
        components::DETAIL,
        task,
        |context: &PageContext<'_>| {
            let id = context
                .param_as::<u32>("id")
                .map_err(|_| PageError::BadRequest)?;
            let title = store::tasks()?
                .into_iter()
                .find(|t| t.id == id)
                .ok_or(PageError::NotFound)?
                .title;
            Ok(
                ClientPageData::new(format!("{title} / Ruvane"), store::snapshot()?)
                    .fallback(native_forms(Some(id))?),
            )
        },
    )
    .unwrap();
    app.client_page(
        PageRoute::new("settings", "/tasks/settings").unwrap(),
        components::SETTINGS,
        zero,
        |_| {
            Ok(ClientPageData::new(
                "Task settings / Ruvane",
                store::snapshot()?,
            ))
        },
    )
    .unwrap();
    app.client_page(
        PageRoute::new("project", "/tasks/project").unwrap(),
        components::PROJECT,
        zero,
        project_page,
    )
    .unwrap();
    app.client_page(
        PageRoute::new("document", "/tasks/document").unwrap(),
        components::PROJECT,
        zero,
        project_page,
    )
    .unwrap();
    app.client_page_document_streaming(
        "document",
        Some(|_| Ok(ruvane::ClientDocument::new("Project / Ruvane"))),
    )
    .unwrap();
    app.client_page(
        PageRoute::new("live", "/tasks/live").unwrap(),
        components::PROJECT,
        zero,
        project_page,
    )
    .unwrap();
    app.client_page_document_streaming(
        "live",
        Some(|_| Ok(ruvane::ClientDocument::new("Project / Ruvane").hydrate_layouts(true))),
    )
    .unwrap();
    app.client_layout(
        PageRoute::new("workspace", "/tasks").unwrap(),
        None,
        components::WORKSPACE,
        zero,
        |_| Ok(ruvane::ClientLayoutData::new(ruvane::ActionData::default())),
    )
    .unwrap();
    app.client_layout(
        PageRoute::new("context", "/tasks/{id}").unwrap(),
        Some("workspace"),
        components::CONTEXT,
        task,
        |_| Ok(ruvane::ClientLayoutData::new(ruvane::ActionData::default())),
    )
    .unwrap();
    for name in ["tasks", "settings", "project", "document", "live"] {
        app.client_page_layout(name, Some("workspace")).unwrap();
    }
    app.client_page_layout("task", Some("context")).unwrap();
    for name in ["workspace", "context"] {
        app.client_layout_cache(name, ruvane::NavigationCachePolicy::public(30).unwrap())
            .unwrap();
    }
    for name in ["home", "tasks", "task", "settings", "project"] {
        app.client_page_cache(name, ruvane::NavigationCachePolicy::public(30).unwrap())
            .unwrap();
    }
    let boundary =
        ruvane::ClientBoundary::new(components::LOADING, components::BOUNDARY_ERROR).unwrap();
    app.client_layout_boundary("workspace", Some(boundary))
        .unwrap();
    app.client_page_boundary("task", Some(boundary)).unwrap();
    app.client_page_boundary("project", Some(boundary)).unwrap();
    app.client_page_streaming("project", true).unwrap();
    app
}
fn project_page(context: &PageContext<'_>) -> Result<ClientPageData, PageError> {
    let query = context.request().unwrap().query_params();
    if query.get("delay") == Some("yes") {
        let delay = std::time::Duration::from_millis(500);
        if let Some(stream) = context.stream_context() {
            stream
                .wait(delay)
                .map_err(|_| PageError::InternalServerError)?;
        } else {
            std::thread::sleep(delay);
        }
    }
    if query.get("unavailable") == Some("yes") {
        return Err(PageError::InternalServerError);
    }
    if query.get("missing") == Some("yes") {
        return Err(PageError::NotFound);
    }
    Ok(ClientPageData::new("Project / Ruvane", store::snapshot()?))
}
pub fn shell(children: Html) -> Html {
    let props = ComponentProps::new(1, InitialState::Number(5)).unwrap();
    let counter = components::create_component(props)
        .unwrap()
        .view()
        .unwrap()
        .hydratable(props)
        .unwrap();
    element(Tag::Main)
        .child(
            element(Tag::Header)
                .child(element(Tag::H1).child("Ruvane application"))
                .child(element(Tag::P).child("Shared Rust pages. No third-party dependencies."))
                .child(application().prefetch_link("home", &[], "Home").unwrap())
                .child(" · ")
                .child(application().prefetch_link("tasks", &[], "Tasks").unwrap())
                .child(" · ")
                .child(ruvane::link(
                    &InternalUrl::new("/stream").unwrap(),
                    "Streaming SSR",
                ))
                .child(" · ")
                .child(ruvane::link(
                    &InternalUrl::new("/tasks/document").unwrap().query("delay", "yes").unwrap(),
                    "Streaming App document",
                ))
                .child(" · ")
                .child(ruvane::link(
                    &InternalUrl::new("/tasks/live").unwrap().query("delay", "yes").unwrap(),
                    "Interactive streamed layouts",
                )),
        )
        .child(counter)
        .child(children)
        .render()
}
fn native_forms(id: Option<u32>) -> Result<Html, PageError> {
    let mut body = element(Tag::Section).child(element(Tag::H2).child("Native forms"));
    if id.is_none() {
        body = body.child(
            PostForm::new(InternalUrl::new("/tasks").unwrap())
                .unwrap()
                .text("title", "Task name", "")
                .unwrap()
                .textarea("notes", "Notes", "")
                .unwrap()
                .submit_label("Add task")
                .unwrap(),
        );
    }
    for task in store::tasks()?
        .into_iter()
        .filter(|t| id.is_none_or(|id| t.id == id))
    {
        let url = InternalUrl::new("/tasks")
            .unwrap()
            .segment(&task.id.to_string())
            .unwrap();
        body = body
            .child(element(Tag::H3).child(task.title.as_str()))
            .child(
                PostForm::new(url.clone().segment("status").unwrap())
                    .unwrap()
                    .checkbox("done", "Completed", task.done)
                    .unwrap()
                    .submit_label("Save status")
                    .unwrap(),
            )
            .child(
                PostForm::new(url.segment("remove").unwrap())
                    .unwrap()
                    .submit_label("Remove task")
                    .unwrap(),
            );
    }
    Ok(body.render())
}
