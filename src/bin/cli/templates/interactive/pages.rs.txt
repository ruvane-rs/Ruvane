use crate::{components, store};
use ruvane::{
    App, ClientPageData, ComponentProps, Html, InitialState, InternalUrl, NavigationMatch,
    PageContext, PageError, PageRoute, PostForm, Tag, element,
};
fn zero(_: &NavigationMatch) -> Result<InitialState, PageError> {
    Ok(InitialState::Number(0))
}
fn task(route: &NavigationMatch) -> Result<InitialState, PageError> {
    let raw = route.param("id").ok_or(PageError::BadRequest)?;
    let id = raw.parse::<u32>().map_err(|_| PageError::BadRequest)?;
    if id == 0 || id.to_string() != raw {
        return Err(PageError::BadRequest);
    }
    Ok(InitialState::Number(id))
}
pub fn application() -> App {
    let mut app = App::new();
    app.components(components::create_component, components::ERROR)
        .unwrap();
    app.client_page(
        PageRoute::new("home", "/").unwrap(),
        components::HOME,
        zero,
        |_| Ok(ClientPageData::new("Home / Ruvane", store::snapshot()?)),
    )
    .unwrap();
    app.client_page(
        PageRoute::new("tasks", "/tasks").unwrap(),
        components::TASKS,
        zero,
        |_| {
            Ok(ClientPageData::new("Tasks / Ruvane", store::snapshot()?)
                .fallback(native_forms(None)?))
        },
    )
    .unwrap();
    app.client_page(
        PageRoute::new("task", "/tasks/{id}").unwrap(),
        components::DETAIL,
        task,
        |context: &PageContext<'_>| {
            let id = context
                .param_as::<u32>("id")
                .map_err(|_| PageError::BadRequest)?;
            let title = store::tasks()?
                .into_iter()
                .find(|t| t.id == id)
                .ok_or(PageError::NotFound)?
                .title;
            Ok(
                ClientPageData::new(format!("{title} / Ruvane"), store::snapshot()?)
                    .fallback(native_forms(Some(id))?),
            )
        },
    )
    .unwrap();
    app.client_page(
        PageRoute::new("settings", "/tasks/settings").unwrap(),
        components::SETTINGS,
        zero,
        |_| {
            Ok(ClientPageData::new(
                "Task settings / Ruvane",
                store::snapshot()?,
            ))
        },
    )
    .unwrap();
    app.client_page(
        PageRoute::new("project", "/tasks/project").unwrap(),
        components::PROJECT,
        zero,
        project_page,
    )
    .unwrap();
    app.client_page(
        PageRoute::new("document", "/tasks/document").unwrap(),
        components::PROJECT,
        zero,
        project_page,
    )
    .unwrap();
    app.client_page_document_streaming(
        "document",
        Some(|_| Ok(ruvane::ClientDocument::new("Project / Ruvane"))),
    )
    .unwrap();
    app.client_page(
        PageRoute::new("live", "/tasks/live").unwrap(),
        components::PROJECT,
        zero,
        project_page,
    )
    .unwrap();
    app.client_page_document_streaming(
        "live",
        Some(|_| Ok(ruvane::ClientDocument::new("Project / Ruvane").hydrate_layouts(true))),
    )
    .unwrap();
    app.client_layout(
        PageRoute::new("workspace", "/tasks").unwrap(),
        None,
        components::WORKSPACE,
        zero,
        |_| Ok(ruvane::ClientLayoutData::new(ruvane::ActionData::default())),
    )
    .unwrap();
    app.client_layout(
        PageRoute::new("context", "/tasks/{id}").unwrap(),
        Some("workspace"),
        components::CONTEXT,
        task,
        |_| Ok(ruvane::ClientLayoutData::new(ruvane::ActionData::default())),
    )
    .unwrap();
    for name in ["tasks", "settings", "project", "document", "live"] {
        app.client_page_layout(name, Some("workspace")).unwrap();
    }
    app.client_page_layout("task", Some("context")).unwrap();
    for name in ["workspace", "context"] {
        app.client_layout_cache(name, ruvane::NavigationCachePolicy::public(30).unwrap())
            .unwrap();
    }
    for name in ["home", "tasks", "task", "settings", "project"] {
        app.client_page_cache(name, ruvane::NavigationCachePolicy::public(30).unwrap())
            .unwrap();
    }
    let boundary =
        ruvane::ClientBoundary::new(components::LOADING, components::BOUNDARY_ERROR).unwrap();
    app.client_boundary(Some(
        ruvane::ClientBoundary::new(components::LOADING, components::GLOBAL_ERROR).unwrap(),
    ))
    .unwrap();
    app.client_layout_boundary("workspace", Some(boundary))
        .unwrap();
    app.client_page_boundary("task", Some(boundary)).unwrap();
    app.client_page_boundary("project", Some(boundary)).unwrap();
    app.client_page_streaming("project", true).unwrap();
    app.client_layout(
        PageRoute::new("globalWorkspace", "/tasks/global").unwrap(),
        None,
        components::WORKSPACE,
        zero,
        |context| {
            if context.request().unwrap().query_params().get("recover") == Some("root")
                && !GLOBAL_ROOT_READY.swap(true, std::sync::atomic::Ordering::Relaxed)
            {
                if let Some(stream) = context.stream_context() {
                    stream
                        .wait(std::time::Duration::from_millis(500))
                        .map_err(|_| PageError::InternalServerError)?;
                }
                return Err(PageError::InternalServerError);
            }
            Ok(ruvane::ClientLayoutData::new(ruvane::ActionData::default()))
        },
    )
    .unwrap();
    app.client_layout(
        PageRoute::new("globalContext", "/tasks/global").unwrap(),
        Some("globalWorkspace"),
        components::CONTEXT,
        zero,
        |_| Ok(ruvane::ClientLayoutData::new(ruvane::ActionData::default())),
    )
    .unwrap();
    app.client_page(
        PageRoute::new("global", "/tasks/global").unwrap(),
        components::PROJECT,
        zero,
        project_page,
    )
    .unwrap();
    app.client_page_layout("global", Some("globalContext"))
        .unwrap();
    app.client_page_streaming("global", true).unwrap();
    app.client_page_document_streaming(
        "global",
        Some(|_| {
            Ok(ruvane::ClientDocument::new("Project / Ruvane")
                .hydrate_layouts(true)
                .hydrate_shell(true))
        }),
    )
    .unwrap();

    app.client_page(
        PageRoute::new("shell", "/tasks/shell").unwrap(),
        components::PROJECT,
        zero,
        project_page,
    )
    .unwrap();
    app.client_page_layout("shell", Some("workspace")).unwrap();
    app.client_page_streaming("shell", true).unwrap();
    app.client_page_document_streaming(
        "shell",
        Some(|context| {
            Ok(ruvane::ClientDocument::new("Project / Ruvane")
                .hydrate_shell(true)
                .hydrate_layouts(
                    context.request().unwrap().query_params().get("layouts") == Some("yes"),
                ))
        }),
    )
    .unwrap();

    app.client_page(
        PageRoute::new("resilient", "/tasks/resilient").unwrap(),
        components::PROJECT,
        zero,
        project_page,
    )
    .unwrap();
    app.client_layout(
        PageRoute::new("resilientContext", "/tasks/resilient").unwrap(),
        Some("workspace"),
        components::CONTEXT,
        zero,
        |context| {
            let query = context.request().unwrap().query_params();
            if query.get("recover") == Some("yes")
                && !LAYOUT_READY.swap(true, std::sync::atomic::Ordering::Relaxed)
            {
                if let Some(stream) = context.stream_context() {
                    stream
                        .wait(std::time::Duration::from_millis(500))
                        .map_err(|_| PageError::InternalServerError)?;
                }
                return Err(PageError::InternalServerError);
            }
            Ok(ruvane::ClientLayoutData::new(ruvane::ActionData::default()))
        },
    )
    .unwrap();
    app.client_page_layout("resilient", Some("resilientContext"))
        .unwrap();
    app.client_page_document_streaming(
        "resilient",
        Some(|_| Ok(ruvane::ClientDocument::new("Project / Ruvane").hydrate_layouts(true))),
    )
    .unwrap();

    app
}
static GLOBAL_ROOT_READY: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
static GLOBAL_PAGE_READY: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
static PARENT_READY: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
static LAYOUT_READY: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
static RECOVERY_READY: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
fn project_page(context: &PageContext<'_>) -> Result<ClientPageData, PageError> {
    let query = context.request().unwrap().query_params();
    if matches!(query.get("delay"), Some("yes" | "long")) {
        let delay = std::time::Duration::from_millis(if query.get("delay") == Some("long") {
            3000
        } else {
            500
        });
        if let Some(stream) = context.stream_context() {
            stream
                .wait(delay)
                .map_err(|_| PageError::InternalServerError)?;
        } else {
            std::thread::sleep(delay);
        }
    }
    if query.get("unavailable") == Some("yes") {
        return Err(PageError::InternalServerError);
    }
    if query.get("missing") == Some("yes") {
        return Err(PageError::NotFound);
    }
    if context.request().unwrap().path() == "/tasks/live"
        && query.get("recover") == Some("yes")
        && !RECOVERY_READY.swap(true, std::sync::atomic::Ordering::Relaxed)
    {
        return Err(PageError::InternalServerError);
    }
    if context.request().unwrap().path() == "/tasks/resilient"
        && query.get("recover") == Some("page")
        && !PARENT_READY.swap(true, std::sync::atomic::Ordering::Relaxed)
    {
        return Err(PageError::InternalServerError);
    }
    if context.request().unwrap().path() == "/tasks/global"
        && query.get("recover") == Some("page")
        && !GLOBAL_PAGE_READY.swap(true, std::sync::atomic::Ordering::Relaxed)
    {
        return Err(PageError::InternalServerError);
    }
    Ok(ClientPageData::new("Project / Ruvane", store::snapshot()?))
}
pub fn shell(children: Html) -> Html {
    let props = ComponentProps::new(1, InitialState::Number(5)).unwrap();
    let counter = components::create_component(props)
        .unwrap()
        .view()
        .unwrap()
        .shell_island("toolbar", props)
        .unwrap();
    element(Tag::Main)
        .child(
            element(Tag::Header)
                .child(element(Tag::H1).child("Ruvane application"))
                .child(element(Tag::P).child("Shared Rust pages. No third-party dependencies."))
                .child(application().prefetch_link("home", &[], "Home").unwrap())
                .child(" · ")
                .child(application().prefetch_link("tasks", &[], "Tasks").unwrap())
                .child(" · ")
                .child(ruvane::link(
                    &InternalUrl::new("/tasks/shell")
                        .unwrap()
                        .query("delay", "long")
                        .unwrap(),
                    "Interactive shell",
                ))
                .child(" · ")
                .child(ruvane::link(
                    &InternalUrl::new("/stream").unwrap(),
                    "Streaming SSR",
                ))
                .child(" · ")
                .child(ruvane::link(
                    &InternalUrl::new("/tasks/document")
                        .unwrap()
                        .query("delay", "yes")
                        .unwrap(),
                    "Streaming App document",
                ))
                .child(" · ")
                .child(ruvane::link(
                    &InternalUrl::new("/tasks/live")
                        .unwrap()
                        .query("delay", "yes")
                        .unwrap(),
                    "Interactive streamed layouts",
                ))
                .child(" · ")
                .child(ruvane::link(
                    &InternalUrl::new("/tasks/live")
                        .unwrap()
                        .query("recover", "yes")
                        .unwrap()
                        .query("delay", "yes")
                        .unwrap(),
                    "Recover a failed page",
                )),
        )
        .child(counter)
        .child(
            element(Tag::P).child(ruvane::link(
                &InternalUrl::new("/tasks/resilient")
                    .unwrap()
                    .query("recover", "yes")
                    .unwrap(),
                "Recover a failed layout",
            )),
        )
        .child(
            element(Tag::P).child(ruvane::link(
                &InternalUrl::new("/tasks/resilient")
                    .unwrap()
                    .query("recover", "page")
                    .unwrap()
                    .query("delay", "yes")
                    .unwrap(),
                "Recover through a parent boundary",
            )),
        )
        .child(
            element(Tag::P).child(ruvane::link(
                &InternalUrl::new("/tasks/global")
                    .unwrap()
                    .query("recover", "root")
                    .unwrap(),
                "Recover a root layout",
            )),
        )
        .child(
            element(Tag::P).child(ruvane::link(
                &InternalUrl::new("/tasks/global")
                    .unwrap()
                    .query("recover", "page")
                    .unwrap()
                    .query("delay", "yes")
                    .unwrap(),
                "Recover an application error",
            )),
        )
        .child(children)
        .render()
}
fn native_forms(id: Option<u32>) -> Result<Html, PageError> {
    let mut body = element(Tag::Section).child(element(Tag::H2).child("Native forms"));
    if id.is_none() {
        body = body.child(
            PostForm::new(InternalUrl::new("/tasks").unwrap())
                .unwrap()
                .text("title", "Task name", "")
                .unwrap()
                .textarea("notes", "Notes", "")
                .unwrap()
                .submit_label("Add task")
                .unwrap(),
        );
    }
    for task in store::tasks()?
        .into_iter()
        .filter(|t| id.is_none_or(|id| t.id == id))
    {
        let url = InternalUrl::new("/tasks")
            .unwrap()
            .segment(&task.id.to_string())
            .unwrap();
        body = body
            .child(element(Tag::H3).child(task.title.as_str()))
            .child(
                PostForm::new(url.clone().segment("status").unwrap())
                    .unwrap()
                    .checkbox("done", "Completed", task.done)
                    .unwrap()
                    .submit_label("Save status")
                    .unwrap(),
            )
            .child(
                PostForm::new(url.segment("remove").unwrap())
                    .unwrap()
                    .submit_label("Remove task")
                    .unwrap(),
            );
    }
    Ok(body.render())
}
