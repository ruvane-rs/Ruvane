#![forbid(unsafe_code)]

use ruvane::ContentBundle;
use std::fs::{self, OpenOptions};
use std::io::{Read, Write};
use std::path::PathBuf;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    println!("cargo:rerun-if-changed=content");
    println!("cargo:rerun-if-env-changed=RUVANE_CONTENT_REVISION");
    let bundle = ContentBundle::read_dir("content")?;
    let source = bundle.rust_module()?;
    let directory =
        PathBuf::from(std::env::var_os("OUT_DIR").ok_or("missing Cargo output directory")?);
    let target = directory.join("ruvane-content.rs");
    if let Ok(metadata) = fs::symlink_metadata(&target) {
        if !metadata.is_file() || metadata.file_type().is_symlink() {
            return Err("expected a regular generated content module".into());
        }
        let mut prefix = Vec::new();
        fs::File::open(&target)?.take(64).read_to_end(&mut prefix)?;
        if !prefix.starts_with(b"// ruvane-content\t1\n") {
            return Err("refusing to replace an unknown Cargo output file".into());
        }
    }
    OpenOptions::new()
        .write(true)
        .create(true)
        .truncate(true)
        .open(target)?
        .write_all(source.as_bytes())?;
    Ok(())
}
